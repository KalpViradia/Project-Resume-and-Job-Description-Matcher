import re
import os

def anonymize_text(text, remove_companies=False):
    text = re.sub(r'[A-Za-z]+\s[A-Za-z]+', 'Candidate', text, count=1)  # Guess name from first line
    text = re.sub(r'\S+@\S+', '[EMAIL_REMOVED]', text)
    text = re.sub(r'(\+?\d[\d\- ]{8,}\d)', '[PHONE_REMOVED]', text)
    text = re.sub(r'(https?://\S+)', '[URL_REMOVED]', text)
    text = re.sub(r'\b(19|20)\d{2}\b', '[YEAR_REMOVED]', text)

    if remove_companies:
        text = re.sub(r'\b([A-Z][a-zA-Z]+ Corp|LLC|Technologies|Labs)\b', '[COMPANY]', text)

    return text


def anonymize_resumes(input_folder="raw_resumes", output_folder="dataset/resumes"):
    for file in os.listdir(input_folder):
        if file.endswith(".txt"):
            with open(os.path.join(input_folder, file), "r", encoding="utf-8") as f:
                text = f.read()

            clean_text = anonymize_text(text, remove_companies=True)

            out_path = os.path.join(output_folder, f"resume_{file}")
            with open(out_path, "w", encoding="utf-8") as out:
                out.write(clean_text)

            print(f"Anonymized → {file}")


if __name__ == "__main__":
    anonymize_resumes()

import pdfplumber
import docx
import os

def extract_text_from_pdf(path):
    text = ""
    with pdfplumber.open(path) as pdf:
        for page in pdf.pages:
            text += page.extract_text() + "\n"
    return text

def extract_text_from_docx(path):
    doc = docx.Document(path)
    return "\n".join([p.text for p in doc.paragraphs])

def extract_folder(input_folder="raw_resumes", output_folder="raw_resumes_txt"):
    os.makedirs(output_folder, exist_ok=True)

    for file in os.listdir(input_folder):
        full_path = os.path.join(input_folder, file)
        out_file = file.rsplit(".", 1)[0] + ".txt"

        if file.endswith(".pdf"):
            text = extract_text_from_pdf(full_path)
        elif file.endswith(".docx"):
            text = extract_text_from_docx(full_path)
        else:
            continue

        with open(os.path.join(output_folder, out_file), "w", encoding="utf-8") as f:
            f.write(text)

        print(f"Extracted → {file}")

if __name__ == "__main__":
    extract_folder()

import csv
import random
import os

def generate_pairs(resume_folder="dataset/resumes", jd_folder="dataset/job_descriptions", output_file="dataset/metadata/match_pairs.csv"):
    resumes = sorted(os.listdir(resume_folder))
    jds = sorted(os.listdir(jd_folder))

    with open(output_file, "w", newline="") as csvfile:

        for resume in resumes:
            jd_match = random.choice(jds)
            writer.writerow([resume, jd_match, 1])  # positive match
            jd_nonmatch = random.choice([jd for jd in jds if jd != jd_match])
            writer.writerow([resume, jd_nonmatch, 0])  # negative match

    print("✅ Match pairs generated!")

    required_skills = random.sample(tech_skills[:6], 3)
    nice_to_have = random.sample(tech_skills[6:], 2)

    jd_text = f"""
Title: {role}
Experience: 1-5 years

Responsibilities:
{chr(10).join(random.sample(responsibilities_templates, 3))}

Required Skills:
{', '.join(required_skills)}

Nice to Have:
{', '.join(nice_to_have)}
"""
    return jd_text.strip()

def generate_jds(n=500, output_folder="dataset/job_descriptions", tech_skills=None):
    os.makedirs(output_folder, exist_ok=True)
    for i in range(1, n+1):
        jd_text = generate_jd(i, tech_skills)
        with open(os.path.join(output_folder, f"jd_{i:03}.txt"), "w") as f:
            f.write(jd_text)
    print(f"✅ Generated {n} synthetic Job Descriptions.")

def generate_resume(index, tech_skills, common_skills):

def generate_match_pairs(resume_folder="dataset/resumes", jd_folder="dataset/job_descriptions", output_file="dataset/metadata/match_pairs.csv"):
    resumes = sorted(os.listdir(resume_folder))
    jds = sorted(os.listdir(jd_folder))

    os.makedirs(os.path.dirname(output_file), exist_ok=True)

    with open(output_file, "w", newline="") as csvfile:
        writer = csv.writer(csvfile)
        writer.writerow(["resume_id", "jd_id", "label"])
        for resume in resumes:
            jd_match = random.choice(jds)
            writer.writerow([resume, jd_match, 1])  # positive match
            jd_nonmatch = random.choice([jd for jd in jds if jd != jd_match])
            writer.writerow([resume, jd_nonmatch, 0])  # negative match

    print("✅ Match pairs CSV generated.")

---

# 1️⃣ **Public / Open Datasets**

These are datasets collected from real sources but cleaned for research. Advantages: realistic patterns, often labeled.

| Dataset                             | Type                      | Notes                                                                                    |
| ----------------------------------- | ------------------------- | ---------------------------------------------------------------------------------------- |
| **Kaggle Job Posting Datasets**     | Job descriptions          | Often pre-processed; some contain location, role, and skills. You can filter tech roles. |
| **O*NET Database**                  | Job descriptions + skills | Extremely structured; used for occupational analysis. Good for skill extraction.         |
| **Resume Datasets on Kaggle**       | Resumes                   | Usually anonymized, but may be small (~200–500 samples).                                 |
| **Indeed / Glassdoor Public Dumps** | Job descriptions          | Some academic datasets contain scraped job postings. Check license.                      |

**Tips:**

* Combine multiple datasets to increase volume.
* Clean for privacy, especially resumes.

---

# 2️⃣ **Web Scraping / APIs (Legal & Safe)**

You can scrape job descriptions or company role descriptions from **publicly available sources**:

* **LinkedIn / Indeed / Glassdoor** (read their ToS carefully!)
* **Company career pages** (public, legal)
* **GitHub jobs API** (deprecated, but there are alternatives like **Adzuna API**)

**Strategy:**

* Scrape only job descriptions, not personal resumes.
* Normalize roles and skills.
* Store anonymized metadata like location, experience.

---

# 3️⃣ **LLMs / Generative Models for Synthetic Data**

You already know about using models like me. Options:

| Model                         | Pros                         | Notes                                                     |
| ----------------------------- | ---------------------------- | --------------------------------------------------------- |
| **ChatGPT / GPT-4 / GPT-5**   | Flexible, realistic language | Can generate multiple variations of resumes & JDs         |
| **Gemini**                    | High-quality output          | Supports structured JSON generation (resume + JD)         |
| **T5 / BART / BLOOM**         | Open-source                  | Can be fine-tuned for generating resumes or rewriting JDs |
| **Text-to-Text Transformers** | Flexible                     | Can paraphrase JDs to increase diversity                  |

**Pro tip:**

* Use structured prompts to generate resumes or JDs in **JSON format**.
* Can mix synthetic with real data for “augmented” datasets.

---

# 4️⃣ **Hybrid Approach (Best Quality + Scale)**

Many researchers combine **real job descriptions + synthetic resumes** or **synthetic JDs + real resumes**:

1. Collect ~500–1000 real JDs from Kaggle / public sources.
2. Generate synthetic resumes using LLMs to match those JDs.
3. Optionally paraphrase JDs using LLMs to create slightly different versions.

✅ Pros:

* Maintains **realistic language** from real JDs.
* Ensures **diverse resumes** without violating privacy.
* Perfect for embedding-based ML pipelines.

---

# 5️⃣ **Other Options for Tech-Specific Data**

* **GitHub profiles / public portfolios**: only scrape public project descriptions.
* **Open-source datasets like Stack Overflow Jobs**: sometimes archived datasets exist with tech role descriptions.
* **Competitions & Hackathons**: some provide datasets of resumes and JD-like descriptions (anonymized).
